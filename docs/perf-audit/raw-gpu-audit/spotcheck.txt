## Script used for section 2 (profile dumps were in /tmp/gpuaudit/profiles/ at the time; run from the dump's directory)
import json
for lab, pas in (("land512_foliage-r1", ("Scene","Tonemap","SSAO")), ("lake512-r2", ("Scene","Tonemap"))):
    d = json.load(open(lab + ".profile.json")); sm = json.load(open(lab + ".summary.json"))
    for p in pas:
        v = sorted(x["ms"] for f in d["frames"] for x in f.get("gpu", []) if x["n"] == p)
        print(lab, p, len(v), v[int(round(.5 * (len(v) - 1)))], sm["gpuPassPercentiles"][p]["p50"])
